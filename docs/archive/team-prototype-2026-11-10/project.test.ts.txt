import { createServer } from "./server";

import {
  PROJECT_PROTOCOL,
  type CreateProjectRequest,
  type CreateProjectTasksRequest,
  type ProjectMemberView,
  type ProjectReceipt,
  type ProjectTaskView,
  type ProjectView,

let server: ReturnType<typeof createServer>,
  db: DatabaseSync,
  root: string,
  databasePath: string;
let creator: AccountSession, member: AccountSession, outsider: AccountSession;
const password = "M1 isolated test password only";
beforeAll(async () => {
  root = await mkdtemp(join(tmpdir(), "praxis-m1-01-"));
  databasePath = join(root, "api.db");
  server = createServer({ databasePath, devices: [] });
  db = new DatabaseSync(databasePath);
  for (const role of ["creator", "member", "outsider"]) {
    const response = await server.inject({
      method: "POST",
      url: "/auth/register",
      payload: {
        email: `${role}@m1.test`,
        displayName: role,
        password,
        deviceId: randomUUID(),
      },
    });
    expect(response.statusCode).toBe(201);
    const account = response.json<AccountSession>();
    if (role === "creator") creator = account;
    else if (role === "member") member = account;
    else outsider = account;
  }
}, 30000);
afterAll(async () => {
  await server?.close();
  db?.close();
  if (root) await rm(root, { recursive: true, force: true });
});
const headers = (session: AccountSession) => ({
  authorization: `Bearer ${session.token}`,
});
function createBody(session = creator): CreateProjectRequest {
  const projectId = randomUUID();
  return {
    protocol: PROJECT_PROTOCOL,
    operationId: randomUUID(),
    deviceId: session.deviceId,
    projectId,
    expectedAccessGeneration: 0,
    name: "project.create",
    payload: {
      projectId,
      name: "协作报名页",
      goal: "",
      targetDate: null,
      timeZone: "Asia/Hong_Kong",
    },
  };
}
const post = (url: string, payload: object, session = creator) =>
  server.inject({ method: "POST", url, headers: headers(session), payload });
const snapshot = (projectId: string, session = creator) =>
  server.inject({
    url: `/v2/projects/${projectId}/snapshot`,
    headers: headers(session),
  });
async function project(session = creator) {
  const body = createBody(session);
  expect((await post("/v2/projects", body, session)).statusCode).toBe(201);
  return body.projectId;
}
function batch(
  projectId: string,
  session = creator,
): CreateProjectTasksRequest {
  return {
    protocol: PROJECT_PROTOCOL,
    operationId: randomUUID(),
    deviceId: session.deviceId,
    projectId,
    expectedAccessGeneration: 1,
    name: "task.create",
    payload: { tasks: [{ id: randomUUID(), title: "完成可验证结果" }] },
  };
}
const submit = (body: ReturnType<typeof batch>, session = creator) =>
  post(`/v2/projects/${body.projectId}/commands`, body, session);
// M1-02 will supply invitation endpoints. Only test setup writes membership rows directly.
function seedMember(
  projectId: string,
  session = member,
  generation = 1,
  state = "active",
) {
  db.prepare("INSERT INTO project_members VALUES(?,?,'member',?,1,?)").run(
    projectId,
    session.user.id,
    state,
    generation,
  );
}
const count = (table: string, projectId: string) =>
  Number(
    db
      .prepare(`SELECT count(*) AS n FROM ${table} WHERE project_id=?`)
      .get(projectId)?.n,
  );

it("aligns implemented response types and protocol with the frozen M0 contract", () => {
  expect(PROJECT_PROTOCOL).toBe(MVP_PROTOCOL);
  expectTypeOf<ProjectView>().toMatchTypeOf<ProjectSnapshot>();
  expectTypeOf<ProjectMemberView>().toMatchTypeOf<ProjectMember>();
  expectTypeOf<ProjectTaskView>().toMatchTypeOf<TaskSnapshot>();
  expectTypeOf<ProjectReceipt>().toMatchTypeOf<CommandReceipt>();
});
it("requires real account sessions for v2, while keeping configured v1 devices working", async () => {
  const token = "m1-legacy-device-only";
  const deviceId = randomUUID();
  const legacy = createServer({
    databasePath: ":memory:",
    devices: [
      {
        ownerId: randomUUID(),
        deviceId,
        tokenHash: createHash("sha256").update(token).digest("hex"),
      },
    ],
  });
  try {
    expect(
      (
        await server.inject({
          method: "POST",
          url: "/v2/projects",
          payload: createBody(),
        })
      ).statusCode,
    ).toBe(401);
    expect(
      (await server.inject({ url: `/v2/projects/${randomUUID()}/snapshot` }))
        .statusCode,
    ).toBe(401);
    expect(
      (
        await legacy.inject({
          url: "/v1/entities/snapshot",
          headers: { authorization: `Bearer ${token}` },
        })
      ).statusCode,
    ).toBe(200);
    expect(
      (
        await legacy.inject({
          method: "POST",
          url: "/v2/projects",
          payload: { ...createBody(), deviceId },
          headers: { authorization: `Bearer ${token}` },
        })
      ).statusCode,
    ).toBe(401);
  } finally {
    await legacy.close();
  }
});
it("creates a project and exactly one creator membership atomically", async () => {
  const body = createBody();
  const response = await post("/v2/projects", body);
  expect(response.statusCode).toBe(201);
  expect(response.headers["cache-control"]).toBe("no-store");
  expect(response.json()).toMatchObject({
    actorId: creator.user.id,
    projectId: body.projectId,
    changedTasks: [],
  });
  const result = await snapshot(body.projectId);
  expect(result.json()).toMatchObject({
    project: {
      name: "协作报名页",
      creatorId: creator.user.id,
      timeZone: "Asia/Hong_Kong",
      version: 1,
      members: [
        {
          userId: creator.user.id,
          role: "creator",
          state: "active",
          accessGeneration: 1,
        },
      ],
    },
    tasks: [],
  });
  expect(count("project_members", body.projectId)).toBe(1);
});
it("rejects unsupported protocols and mismatched command project IDs without writes", async () => {
  const create = createBody();
  expect(
    (
      await post("/v2/projects", {
        ...create,
        protocol: "unsupported-protocol",
      })
    ).json(),
  ).toEqual({ error: "PROTOCOL_UNSUPPORTED" });
  expect(
    db
      .prepare("SELECT 1 FROM shared_projects WHERE id=?")
      .get(create.projectId),
  ).toBeUndefined();
  const projectId = await project();
  const body = batch(projectId);
  expect(
    (
      await post(`/v2/projects/${projectId}/commands`, {
        ...body,
        protocol: "unsupported-protocol",
      })
    ).json(),
  ).toEqual({ error: "PROTOCOL_UNSUPPORTED" });
  expect(
    (
      await post(`/v2/projects/${projectId}/commands`, {
        ...body,
        projectId: randomUUID(),
      })
    ).statusCode,
  ).toBe(400);
  expect(count("project_tasks", projectId)).toBe(0);
});
it.each([
  "blank-name",
  "invalid-date",
  "invalid-zone",
  "forged-creator",
  "mismatched-id",
])("rejects %s without creating any project", async (variant) => {
  const body = createBody();
  const payload: Record<string, unknown> = { ...body.payload };
  if (variant === "blank-name") payload.name = "   ";
  if (variant === "invalid-date") payload.targetDate = "2026-02-30";
  if (variant === "invalid-zone") payload.timeZone = "Invalid/Zone";
  if (variant === "forged-creator") payload.creatorId = outsider.user.id;
  if (variant === "mismatched-id") payload.projectId = randomUUID();
  expect((await post("/v2/projects", { ...body, payload })).statusCode).toBe(
    400,
  );
  expect(
    db.prepare("SELECT 1 FROM shared_projects WHERE id=?").get(body.projectId),
  ).toBeUndefined();
});
it("rejects forged devices and identity fields before any writes", async () => {

  ).toBe(403);
  expect(
    (await post("/v2/projects", { ...body, actorId: outsider.user.id }))

  expect(

});
it("allows real members to read and create tasks with trusted creator and default assignee", async () => {
  const projectId = await project();
  seedMember(projectId);
  const body = batch(projectId, member);
  expect((await submit(body, member)).statusCode).toBe(200);
  const response = await snapshot(projectId, member);
  expect(response.statusCode).toBe(200);
  expect(response.json().tasks).toMatchObject([
    {
      scope: { kind: "project", projectId },
      createdBy: member.user.id,
      assigneeId: member.user.id,
      status: "todo",
      notes: "",
      deadline: null,
      version: 1,
    },
  ]);
  const assigned = {
    ...batch(projectId, member),
    payload: {
      tasks: [
        { id: randomUUID(), title: "明确分配", assigneeId: creator.user.id },
      ],
    },
  };
  expect((await submit(assigned, member)).statusCode).toBe(200);
  expect(
    (await snapshot(projectId))
      .json()
      .tasks.find(
        (task: { id: string }) => task.id === assigned.payload.tasks[0]!.id,
      ),
  ).toMatchObject({ createdBy: member.user.id, assigneeId: creator.user.id });
});

});

it("rolls back an entire task batch when a later database write fails", async () => {
  const projectId = await project();
  const secondId = randomUUID();
  const before = (await snapshot(projectId)).json();
  const body = {
    ...batch(projectId),
    payload: {
      tasks: [
        { id: randomUUID(), title: "先写第一项" },
        { id: secondId, title: "第二项被故障拒绝" },
      ],
    },
  };
  db.exec(
    `CREATE TRIGGER m1_fail_task BEFORE INSERT ON project_tasks WHEN NEW.id='${secondId}' BEGIN SELECT RAISE(ABORT,'synthetic failure'); END`,
  );
  try {
    expect((await submit(body)).statusCode).toBe(500);
  } finally {
    db.exec("DROP TRIGGER m1_fail_task");
  }
  expect(count("project_tasks", projectId)).toBe(0);
  expect(count("shared_project_events", projectId)).toBe(1);
  expect((await snapshot(projectId)).json().project).toEqual(before.project);
  expect(
    db
      .prepare("SELECT 1 FROM shared_command_receipts WHERE operation_id=?")
      .get(body.operationId),
  ).toBeUndefined();
  expect((await submit(body)).statusCode).toBe(200);
  expect(count("project_tasks", projectId)).toBe(2);
});

  } finally {
    db.exec("DROP TRIGGER m1_fail_receipt");
  }
  expect(
    db.prepare("SELECT 1 FROM shared_projects WHERE id=?").get(body.projectId),
  ).toBeUndefined();
  expect(count("project_members", body.projectId)).toBe(0);

});

it("replays identical requests across devices and rejects changed operation contents", async () => {
  const body = createBody();
  const created = await post("/v2/projects", body);
  expect((await post("/v2/projects", body)).json()).toEqual(created.json());
  expect(
    (
      await post("/v2/projects", {
        ...body,
        payload: { ...body.payload, name: "不同名称" },
      })
    ).json(),
  ).toEqual({ error: "OPERATION_CHANGED" });
  const tasks = batch(body.projectId);
  const first = await submit(tasks);
  const login = await server.inject({
    method: "POST",
    url: "/auth/login",
    payload: { email: creator.user.email, password, deviceId: randomUUID() },
  });
  const secondDevice = login.json<AccountSession>();
  expect(
    (
      await submit({ ...tasks, deviceId: secondDevice.deviceId }, secondDevice)
    ).json(),
  ).toEqual(first.json());
  expect(
    (
      await submit({
        ...tasks,
        payload: { tasks: [{ ...tasks.payload.tasks[0]!, title: "改了草稿" }] },
      })
    ).json(),
  ).toEqual({ error: "OPERATION_CHANGED" });
  expect(count("project_tasks", body.projectId)).toBe(1);
  expect(
    (
      await server.inject({
        url: `/v2/commands/${tasks.operationId}`,
        headers: headers(secondDevice),
      })
    ).json(),
  ).toEqual(first.json());
  expect(
    (
      await server.inject({
        url: `/v2/commands/${tasks.operationId}`,
        headers: headers(outsider),
      })
    ).statusCode,
  ).toBe(403);
});
it("preserves shared projects, tasks, receipts and old personal data across server reopen", async () => {
  const projectId = await project();
  const tasks = batch(projectId);
  const applied = (await submit(tasks)).json();
  const before = (await snapshot(projectId)).json();
  const privateBefore = (
    await server.inject({
      url: "/v1/entities/snapshot",
      headers: headers(creator),
    })
  ).json().entities;
  await server.close();
  server = createServer({ databasePath, devices: [] });
  const after = (await snapshot(projectId)).json();
  expect(after.project).toEqual(before.project);
  expect(after.tasks).toEqual(before.tasks);
  expect((await submit(tasks)).json()).toEqual(applied);
  expect(
    (
      await server.inject({
        url: "/v1/entities/snapshot",
        headers: headers(creator),
      })
    ).json().entities,
  ).toEqual(privateBefore);
});
